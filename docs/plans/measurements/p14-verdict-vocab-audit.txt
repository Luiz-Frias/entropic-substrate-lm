Command reproducing this file
-----------------------------
Run from the repo root at the start tip, with the project venv:

    PYTHONPATH=src .venv/bin/python audit.py            # the AUDIT SCRIPT below
    PYTHONPATH=src .venv/bin/python control.py <dir>    # the CONTROL BUILDER below
    PYTHONPATH=src .venv/bin/python audit.py --control <dir>

No script file is committed (the p14-text-trim-audit.txt precedent: Allowed
files admits only this .txt); the script text is quoted verbatim below. The
instrument: a node is an ARTIFACT when it is a dict whose schema_version is
d1-verdict/v2 or d1-verdict/v3 (the two schema versions the production
validator accepts, d1_validator.py:87). Each artifact's vocabulary-bearing sections are read back
through the production parsers `_parse_contrasts`, `_parse_clauses_by_arm`,
`_parse_integrity` (ladder/file/d1_validator.py) and `parse_review`
(ladder/file/d1_decision_validation.py), and every value of the seven
vocabulary fields is counted from the raw payload and compared against the
legacy sets independently of the parser. Strings under keys ending
`_base64` are decoded and walked (decision contracts travel that way).
A decode failure on a claimed-decodable file is reported as UNREADABLE,
never dropped.

    """FOR-1573 persisted D1 verdict vocabulary audit. Not committed."""
    from __future__ import annotations

    import argparse
    import base64
    import collections
    import json
    import subprocess
    import sys
    from pathlib import Path

    from entropic_substrate.kernel.verdict.d1_types import (
        CLAUSE_STATUS,
        CONTRAST_DECISIONS,
        CONTRAST_KINDS,
        EVIDENCE_CLASSES,
        D1VerdictValidationError,
    )
    from entropic_substrate.ladder.file.d1_decision_validation import parse_review
    from entropic_substrate.ladder.file.d1_validator import (
        _parse_clauses_by_arm,
        _parse_contrasts,
        _parse_integrity,
    )

    SCHEMAS = {"d1-verdict/v2", "d1-verdict/v3"}
    MARKERS = (b"d1-verdict/v", b"clauses_by_arm", b'"contrast_kind"')
    ADJUDICATION_STATUS = {"OPEN", "RESOLVED", "NOT_REQUIRED"}
    DISPOSITION_STATUS = {"OPEN", "RESOLVED"}
    FIELDS = (
        "contrast_kind",
        "evidence_class",
        "contrast_decision",
        "clause_status",
        "integrity_status",
        "adjudication_status",
        "disposition_status",
    )
    LEGAL = {
        "contrast_kind": CONTRAST_KINDS,
        "evidence_class": EVIDENCE_CLASSES,
        "contrast_decision": CONTRAST_DECISIONS,
        "clause_status": CLAUSE_STATUS,
        "integrity_status": CLAUSE_STATUS,
        "adjudication_status": ADJUDICATION_STATUS,
        "disposition_status": DISPOSITION_STATUS,
    }


    class Unreadable(Exception):
        """A claimed-decodable file or base64 field could not be decoded."""


    def is_artifact(node: object) -> bool:
        return type(node) is dict and node.get("schema_version") in SCHEMAS


    def is_signature(node: object) -> bool:
        return type(node) is dict and "clauses_by_arm" in node and "contrasts" in node


    def tally(node: dict, counts: dict[str, collections.Counter]) -> None:
        for contrast in node.get("contrasts", []) if type(node.get("contrasts")) is list else []:
            if type(contrast) is dict:
                counts["contrast_kind"][json.dumps(contrast.get("contrast_kind"))] += 1
                counts["evidence_class"][json.dumps(contrast.get("evidence_class"))] += 1
                counts["contrast_decision"][json.dumps(contrast.get("decision"))] += 1
        by_arm = node.get("clauses_by_arm")
        if type(by_arm) is dict:
            for clauses in by_arm.values():
                if type(clauses) is dict:
                    for clause in clauses.values():
                        if type(clause) is dict:
                            counts["clause_status"][json.dumps(clause.get("status"))] += 1
        integrity = node.get("integrity")
        if type(integrity) is dict:
            counts["integrity_status"][json.dumps(integrity.get("status"))] += 1
        review = node.get("review")
        adjudication = review.get("adjudication") if type(review) is dict else None
        if type(adjudication) is dict:
            counts["adjudication_status"][json.dumps(adjudication.get("status"))] += 1
            for item in adjudication.get("dispositions", []) if type(
                adjudication.get("dispositions")
            ) is list else []:
                if type(item) is dict:
                    counts["disposition_status"][json.dumps(item.get("status"))] += 1


    def real_parse(node: dict) -> str:
        """Read the vocabulary-bearing sections back through the production parsers."""
        try:
            _parse_contrasts(node.get("contrasts"), allow_gaps=True)
            _parse_clauses_by_arm(node.get("clauses_by_arm"))
            _parse_integrity(node.get("integrity"))
            parse_review(node.get("review"))
        except D1VerdictValidationError as exc:
            return f"REJECTED {exc.code} at {exc.path}"
        except Exception as exc:  # reported, never swallowed
            return f"PARSER-ERROR {type(exc).__name__}: {exc}"
        return "PARSED"


    def walk(node, path, source, found):
        if is_artifact(node) or is_signature(node):
            found.append((source, path, "artifact" if is_artifact(node) else "signature", node))
        if type(node) is dict:
            for key, value in node.items():
                child = f"{path}.{key}" if path else str(key)
                if type(value) is str and key.endswith("_base64"):
                    try:
                        raw = base64.b64decode(value, validate=True)
                    except Exception as error:
                        raise Unreadable(f"{source}:{child}: invalid base64: {error}") from error
                    try:
                        decoded = json.loads(raw.decode("utf-8"))
                    except (UnicodeDecodeError, json.JSONDecodeError):
                        continue
                    walk(decoded, f"{child}(base64)", source, found)
                else:
                    walk(value, child, source, found)
        elif type(node) is list:
            for index, item in enumerate(node):
                walk(item, f"{path}[{index}]", source, found)


    def magic_json(paths: list[Path]) -> set[Path]:
        if not paths:
            return set()
        proc = subprocess.run(
            ["file", "-f", "-"],
            input="\n".join(str(p) for p in paths),
            capture_output=True,
            text=True,
            check=True,
        )
        hits = set()
        for line in proc.stdout.splitlines():
            name, _, desc = line.rpartition(": ")
            if "JSON data" in desc:
                hits.add(Path(name))
        return hits


    def scan(label: str, files: list[Path], out: list[str]) -> dict:
        claimed = sorted(
            {p for p in files if p.suffix in (".json", ".jsonl")} | magic_json(files)
        )
        found, unreadable, markers = [], [], []
        for path in files:
            try:
                blob = path.read_bytes()
            except OSError as exc:
                unreadable.append(f"{path}: {exc}")
                continue
            if any(m in blob for m in MARKERS):
                markers.append(str(path))
        for path in claimed:
            try:
                text = path.read_text(encoding="utf-8")
            except (OSError, UnicodeDecodeError) as exc:
                unreadable.append(f"{path}: {exc}")
                continue
            docs = []
            try:
                if path.suffix == ".jsonl":
                    for number, line in enumerate(text.splitlines(), 1):
                        if line.strip():
                            docs.append((f"{path}:{number}", json.loads(line)))
                else:
                    try:
                        docs.append((str(path), json.loads(text)))
                    except json.JSONDecodeError:
                        for number, line in enumerate(text.splitlines(), 1):
                            if line.strip():
                                docs.append((f"{path}:{number}", json.loads(line)))
            except json.JSONDecodeError as exc:
                unreadable.append(f"{path}: JSONDecodeError {exc}")
                continue
            try:
                for source, doc in docs:
                    walk(doc, "", source, found)
            except Unreadable as exc:
                unreadable.append(str(exc))
        counts = {name: collections.Counter() for name in FIELDS}
        verdicts = []
        for source, path, kind, node in found:
            tally(node, counts)
            verdicts.append((source, path, kind, real_parse(node)))
        out.append(f"[{label}]")
        out.append(f"  files in scan set                 : {len(files)}")
        out.append(f"  claimed decodable (json/jsonl/magic): {len(claimed)}")
        out.append(f"  files containing a D1 marker byte string: {len(markers)}")
        for item in markers:
            out.append(f"    marker file: {item}")
        out.append(f"  D1 verdict artifacts (schema_version d1-verdict/v2|v3): "
                   f"{sum(1 for v in verdicts if v[2] == 'artifact')}")
        out.append(f"  schemaless signature hits (clauses_by_arm + contrasts): "
                   f"{sum(1 for v in verdicts if v[2] == 'signature')}")
        for source, path, kind, result in verdicts:
            out.append(f"    {kind} {source} {path or '<root>'} -> {result}")
        for name in FIELDS:
            entries = []
            bad = 0
            for value, number in sorted(counts[name].items()):
                raw = json.loads(value)
                legal = type(raw) is str and raw in LEGAL[name]
                bad += 0 if legal else number
                entries.append(f"{value}x{number}{'' if legal else ' OUT-OF-VOCABULARY'}")
            out.append(f"  {name}: {', '.join(entries) if entries else '(none observed)'}"
                       f" ; out-of-vocabulary={bad}")
        out.append(f"  UNREADABLE: {len(unreadable)}")
        for item in unreadable:
            out.append(f"    {item}")
        return {"artifacts": len(verdicts), "unreadable": len(unreadable)}


    def main() -> None:
        parser = argparse.ArgumentParser()
        parser.add_argument("--control", type=Path)
        args = parser.parse_args()
        out: list[str] = []
        if args.control:
            files = sorted(p for p in args.control.rglob("*") if p.is_file())
            scan(f"CONTROL {args.control}", files, out)
            print("\n".join(out))
            return
        tracked = subprocess.run(
            ["git", "ls-files"], capture_output=True, text=True, check=True
        ).stdout.splitlines()
        scan("TRACKED TREE (git ls-files)", [Path(p) for p in tracked if Path(p).is_file()], out)
        results = Path("results")
        pointer_outs = {
            "captured": "results/captured.dvc",
            "h-image-publication-79a4be2": "results/h-image-publication-79a4be2.dvc",
            "h-image-publication-88d1a25": "results/h-image-publication-88d1a25.dvc",
            "h-image-publication-9a6300a": "results/h-image-publication-9a6300a.dvc",
            "h-image-publication-b576df4": "results/h-image-publication-b576df4.dvc",
        }
        for name in pointer_outs:
            root = results / name
            scan(f"DVC OUTPUT results/{name}", sorted(p for p in root.rglob("*") if p.is_file()), out)
        scratch = sorted(
            entry for entry in results.iterdir()
            if entry.is_dir() and entry.name not in pointer_outs
        )
        for entry in scratch:
            scan(f"UNTRACKED SCRATCH results/{entry.name} (classified by name: source/scratch copy)",
                 sorted(p for p in entry.rglob("*") if p.is_file()), out)
        loose = sorted(e for e in results.iterdir() if e.is_file() and e.suffix != ".dvc")
        scan("LOOSE FILES results/* (non-pointer)", loose, out)
        scan(".dvc/cache", sorted(p for p in Path(".dvc/cache").rglob("*") if p.is_file()), out)
        print("\n".join(out))


    if __name__ == "__main__":
        sys.exit(main())
